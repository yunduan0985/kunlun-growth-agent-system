import json
import os

log_path = "/Users/dasean/.gemini/antigravity/brain/2fd39c36-f2f1-4a9c-b0fe-7e41d7e46122/.system_generated/logs/transcript_full.jsonl"
out_dir = "scratch/extracted_chunks_older"
os.makedirs(out_dir, exist_ok=True)

print(f"
=====================
 Extracting code chunks from older transcript: {log_path}...")
with open(log_path, 'r', encoding='utf-8') as f:
    for line in f:
        try:
            step = json.loads(line)
            tool_calls = step.get('tool_calls', [])
            for call in tool_calls:
                name = call.get('name')
                args = call.get('args', {})
                if 'replace_file_content' in name or 'multi_replace_file_content' in name:
                    step_idx = step.get('step_index')
                    print(f"
=====================
 Step {step_idx}: {name}")
                    
                    # For replace_file_content
                    if 'ReplacementContent' in args:
                        chunk_path = os.path.join(out_dir, f"step_{step_idx}_replace.js")
                        with open(chunk_path, 'w', encoding='utf-8') as out:
                            out.write(args['ReplacementContent'])
                        print(f"   Saved chunk to {chunk_path}")
                    
                    # For multi_replace_file_content
                    if 'ReplacementChunks' in args:
                        for idx, chunk in enumerate(args['ReplacementChunks']):
                            chunk_path = os.path.join(out_dir, f"step_{step_idx}_multi_{idx}.js")
                            with open(chunk_path, 'w', encoding='utf-8') as out:
                                out.write(chunk['ReplacementContent'])
                            print(f"   Saved chunk to {chunk_path}")
        except Exception as e:
            pass
